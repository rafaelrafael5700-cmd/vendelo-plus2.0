// --- PUNTO DE ENTRADA GET ---
function doGet(e) {
  var action = e.parameter.accion;
  try {
    if (action === "login") return response(login(e.parameter.usuario, e.parameter.password));
    if (action === "registrarNuevoCliente") {
      return response(registrarNuevoCliente(e.parameter.id, e.parameter.nombre, e.parameter.correo, e.parameter.password));
    }
    if (action === "obtenerClientes") return response(obtenerClientes());
    if (action === "obtenerPlanes") return response(obtenerPlanes());
    if (action === "obtenerConfigMaestro") return response(obtenerConfigMaestro());
    if (["bloquear", "desbloquear", "eliminar", "aplicarPlan"].indexOf(action) > -1) {
      return response(gestionarUsuario(e.parameter));
    }
    if (e.parameter.id) return response(validarLicencia(e.parameter.id));
  } catch (err) {
    return response({ok: false, error: "Error Maestro (GET): " + err.toString()});
  }
  return response({ok: true, msg: "MASTER VÉNDELO+ ONLINE - SECURITY ACTIVE"});
}

// --- PUNTO DE ENTRADA POST ---
function doPost(e) {
  try {
    var body;
    // Intentar obtener body de JSON o de parámetros (URLSearchParams)
    if (e.postData && e.postData.contents) {
      try {
        body = JSON.parse(e.postData.contents);
      } catch (err) {
        body = e.parameter;
      }
    } else {
      body = e.parameter;
    }

    var action = body.accion || e.parameter.accion;
    var sheetId = body.sheetId || e.parameter.sheetId;

    if (action === "actualizarPlanes") {
      var planesData = body.dataPlanes || e.parameter.dataPlanes;
      return response(actualizarPlanes(typeof planesData === 'string' ? JSON.parse(planesData) : planesData));
    }
    
    // Acción admin: guardar config global del maestro (sin sheetId de cliente)
    if (action === "guardarConfigMaestro") {
      return response(guardarConfigMaestroFn(body.clave, body.valor));
    }

    if (!sheetId) throw new Error("Falta ID de base de datos");
    
    // --- VALIDACIÓN DE SEGURIDAD EN TIEMPO REAL ---
    var status = verificarLicenciaPorSheetId(sheetId);
    if (!status.permitido) {
      return response({ ok: false, error: "LICENCIA RESTRINGIDA: Tu cuenta está bloqueada o vencida. Contacta a soporte." });
    }

    var ss = SpreadsheetApp.openById(sheetId);
    var result = handleAction(ss, action, body);
    return response(result);
  } catch (error) {
    return response({ok: false, error: "Error Maestro (POST): " + error.toString()});
  }
}

function response(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

// --- HELPER: VERIFICAR LICENCIA POR SHEET ID ---
function verificarLicenciaPorSheetId(sheetId) {
  var sheet = getMasterSheet("Clientes");
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][4]) === String(sheetId)) {
      var hoy = new Date();
      var vencimiento = new Date(data[i][8]);
      var vencido = vencimiento < hoy;
      var bloqueado = String(data[i][6]).toUpperCase() === "BLOQUEADO";
      return { permitido: !vencido && !bloqueado, plan: data[i][5], status: data[i][6] };
    }
  }
  return { permitido: false };
}

// --- LÓGICA CRM (Ruteo Multi-Tenant) ---

function handleAction(ss, action, params) {
  if (action === 'cargarTodo') return cargarTodo(ss);
  if (action === 'guardarProducto') return guardarProducto(ss, params.producto, params.oldId);
  if (action === 'eliminarProducto') return eliminarProducto(ss, params.id);
  if (action === 'guardarMovimiento') return guardarMovimiento(ss, params.movimiento);
  if (action === 'actualizarStock') return actualizarStock(ss, params.id, params.cantidad, params.costo);
  if (action === 'registrarUsuario') return registrarUsuarioInterno(ss, params.usuario);
  
  if (action === 'guardarContacto') return guardarContacto(ss, params.contacto);
  if (action === 'eliminarContacto') return eliminarContacto(ss, params.id);
  if (action === 'guardarConfig') return guardarConfig(ss, params.clave, params.valor);

  return {ok: false, error: "Acción desconocida en el Maestro."};
}

function login(u, p) {
  var sheet = getMasterSheet("Clientes");
  var data = sheet.getDataRange().getValues();
  var userBusqueda = String(u).split('@')[0].toLowerCase();
  for (var i = 1; i < data.length; i++) {
    var dbId = String(data[i][0]).toLowerCase(); 
    var dbCorreo = String(data[i][2]).toLowerCase(); 
    if ((dbId === userBusqueda || dbCorreo === String(u).toLowerCase()) && String(data[i][3]) === String(p)) {
      var hoy = new Date();
      var vencimiento = new Date(data[i][8]);
      var vencido = vencimiento < hoy;
      var bloqueado = String(data[i][6]).toUpperCase() === "BLOQUEADO";

      return {
        ok: true,
        usuario: { 
          id: data[i][0], 
          nombre: data[i][1], 
          correo: data[i][2], 
          sheetId: data[i][4], 
          plan: data[i][5], 
          status: (vencido || bloqueado) ? "BLOQUEADO" : data[i][6],
          permitido: !vencido && !bloqueado 
        }
      };
    }
  }
  return {ok: false, error: "Usuario o contraseña incorrectos"};
}

function registrarNuevoCliente(id, nombre, correo, password) {
  var sheet = getMasterSheet("Clientes");
  var data = sheet.getDataRange().getValues();
  var idN = String(id).trim().toLowerCase();
  var correoN = String(correo).trim().toLowerCase();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).toLowerCase() === idN) return {ok: false, error: "El nombre de usuario ya existe."};
    if (String(data[i][2]).toLowerCase() === correoN) return {ok: false, error: "El correo ya está registrado."};
  }
  var folder = DriveApp.getFolderById(CARPETA_ID);
  var copia = DriveApp.getFileById(PLANTILLA_ID).makeCopy("DB_" + nombre, folder);
  var sid = copia.getId();
  inicializar(SpreadsheetApp.openById(sid));
  
  // REGALO DE 30 DÍAS PLAN PRO PARA USUARIOS NUEVOS
  var vencimiento = new Date(); 
  vencimiento.setDate(vencimiento.getDate() + 30);
  
  sheet.appendRow([id, nombre, correo, password, sid, "PRO", "ACTIVO", new Date(), vencimiento]);
  return {ok: true, sheetId: sid};
}

function validarLicencia(id) {
  var data = getMasterSheet("Clientes").getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(id)) {
      var hoy = new Date();
      var vencimiento = new Date(data[i][8]);
      var vencido = vencimiento < hoy;
      var bloqueado = String(data[i][6]).toUpperCase() === "BLOQUEADO";
      return { 
        permitido: !vencido && !bloqueado, 
        plan: data[i][5], 
        status: data[i][6],
        vencimiento: data[i][8]
      };
    }
  }
  return { permitido: false };
}

function gestionarUsuario(p) {
  var sheet = getMasterSheet("Clientes");
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(p.id)) {
      if (p.accion === "bloquear") sheet.getRange(i+1, 7).setValue("Bloqueado");
      if (p.accion === "desbloquear") sheet.getRange(i+1, 7).setValue("Activo");
      if (p.accion === "aplicarPlan") {
        var v = new Date(); v.setDate(v.getDate() + parseInt(p.dias));
        sheet.getRange(i+1, 6).setValue(p.nombrePlan);
        sheet.getRange(i+1, 9).setValue(v);
        sheet.getRange(i+1, 7).setValue("Activo"); // Al aplicar plan se desbloquea
      }
      if (p.accion === "eliminar") { try { DriveApp.getFileById(data[i][4]).setTrashed(true); } catch(e){} sheet.deleteRow(i+1); }
      return {ok: true};
    }
  }
  return {ok: false, error: "Usuario no encontrado"};
}

// ... (Resto de funciones de CRM se mantienen igual)
function getMasterSheet(name) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName(name) || ss.insertSheet(name);
}

function inicializar(ss) {
  var tablas = [
    {n: "Productos", h: ["id", "nombre", "costo", "venta", "stock", "stockMin", "stockMax", "unidad", "bloque", "estante", "img", "proveedor", "categoria", "barcode"]},
    {n: "Movimientos", h: ["id", "fecha", "fechaISO", "montoTotal", "tipo", "metodo", "items", "mixto", "cliente", "vencimiento", "estado", "proveedor"]},
    {n: "Directorio", h: ["id", "nombre", "tel", "tipo", "saldo"]}, // <-- AÑADIDO: saldo
    {n: "Configuracion", h: ["clave", "valor"]},
    {n: "Usuarios", h: ["nombre", "correo", "usuario", "password", "departamento"]}
  ];
  tablas.forEach(t => {
    var s = ss.getSheetByName(t.n);
    if (!s) { 
      s = ss.insertSheet(t.n); 
      s.appendRow(t.h); 
    }
  });
}

function cargarTodo(ss) {
  var res = { ok: true, productos: [], movimientos: [], directorio: [], config: {} };
  var mappings = {"Productos": "productos", "Movimientos": "movimientos", "Directorio": "directorio", "Configuracion": "config"};
  for (var n in mappings) {
    var s = ss.getSheetByName(n);
    if (!s || s.getLastRow() < 2) continue;
    var rows = s.getDataRange().getValues();
    var h = rows.shift();
    var key = mappings[n];
    if (key === "config") rows.forEach(r => res.config[r[0]] = r[1]);
    else {
      res[key] = rows.map(r => {
        var o = {};
        h.forEach((head, i) => {
          if (head === "items" || head === "mixto") {
            try { o[head] = r[i] ? JSON.parse(r[i]) : []; } catch(e) { o[head] = r[i]; }
          } else { o[head] = r[i]; }
        });
        return o;
      });
      if (key === "movimientos") res[key].reverse();
    }
  }
  return res;
}

function guardarProducto(ss, p, oldId) {
  if (typeof p === 'string') p = JSON.parse(p);
  var s = ss.getSheetByName("Productos");
  var d = s.getDataRange().getValues();
  var idx = -1;
  var sid = String(oldId || p.id);
  for (var i = 1; i < d.length; i++) { if (String(d[i][0]) === sid) { idx = i + 1; break; } }
  var row = [p.id, p.nombre, p.costo, p.venta, p.stock, p.stockMin, p.stockMax, p.unidad, p.bloque, p.estante, p.img, p.proveedor, p.categoria, p.barcode];
  if (idx > 0) s.getRange(idx, 1, 1, row.length).setValues([row]);
  else s.appendRow(row);
  return {ok: true};
}

function guardarMovimiento(ss, m) {
  if (typeof m === 'string') m = JSON.parse(m);
  var s = ss.getSheetByName("Movimientos");
  s.appendRow([m.id, m.fecha, m.fechaISO, m.montoTotal, m.tipo, m.metodo, JSON.stringify(m.items), JSON.stringify(m.mixto), m.cliente, m.vencimiento, m.estado, m.proveedor]);
  return {ok: true};
}

function actualizarStock(ss, id, cant, costo) {
  var s = ss.getSheetByName("Productos");
  var d = s.getDataRange().getValues();
  for (var i = 1; i < d.length; i++) {
    if (String(d[i][0]) === String(id)) {
      var nS = (parseFloat(d[i][4]) || 0) + parseFloat(cant);
      s.getRange(i+1, 5).setValue(nS);
      if (costo) s.getRange(i+1, 3).setValue(parseFloat(costo));
      return {ok: true, newStock: nS};
    }
  }
  return {ok: false};
}

function registrarUsuarioInterno(ss, u) {
  if (typeof u === 'string') u = JSON.parse(u);
  var s = ss.getSheetByName("Usuarios");
  var d = s.getDataRange().getValues();
  for (var i = 1; i < d.length; i++) {
    if (String(d[i][2]).toLowerCase() === String(u.usuario).toLowerCase()) return {ok: false, error: "Usuario interno duplicado"};
  }
  s.appendRow([u.nombre, u.correo, u.usuario, u.password, u.departamento]);
  return {ok: true};
}

function guardarContacto(ss, contacto) {
  if (typeof contacto === 'string') contacto = JSON.parse(contacto);
  var s = ss.getSheetByName("Directorio");
  if (!s) return {ok: false, error: "Hoja Directorio no encontrada"};
  
  // AÑADIDO: Validar que la cabecera 'saldo' exista (retrocompatibilidad para bases de datos viejas)
  var hCols = s.getRange(1, 1, 1, s.getLastColumn() || 1).getValues()[0];
  if (hCols.indexOf("saldo") === -1) {
    s.getRange(1, hCols.length + 1).setValue("saldo");
  }

  var d = s.getDataRange().getValues();
  var idx = -1;
  var sid = String(contacto.id);
  for (var i = 1; i < d.length; i++) { if (String(d[i][0]) === sid) { idx = i + 1; break; } }
  
  // AÑADIDO: Incluir contacto.saldo (por defecto 0) como 5to elemento del array
  var row = [contacto.id, contacto.nombre, contacto.tel, contacto.tipo, contacto.saldo || 0];
  
  if (idx > 0) s.getRange(idx, 1, 1, row.length).setValues([row]);
  else s.appendRow(row);
  return {ok: true};
}

function eliminarContacto(ss, id) {
  var s = ss.getSheetByName("Directorio");
  var d = s.getDataRange().getValues();
  for (var i = 1; i < d.length; i++) {
    if (String(d[i][0]) === String(id)) { s.deleteRow(i + 1); return {ok: true}; }
  }
  return {ok: false, error: "Contacto no encontrado"};
}

function guardarConfig(ss, clave, valor) {
  var s = ss.getSheetByName("Configuracion");
  if (!s) return {ok: false, error: "Hoja Configuracion no encontrada"};
  var d = s.getDataRange().getValues();
  var idx = -1;
  for (var i = 1; i < d.length; i++) { if (String(d[i][0]) === String(clave)) { idx = i + 1; break; } }
  if (idx > 0) s.getRange(idx, 2).setValue(valor);
  else s.appendRow([clave, valor]);
  return {ok: true};
}

// ===== CONFIG GLOBAL DEL MAESTRO (WhatsApp soporte, etc.) =====

function obtenerConfigMaestro() {
  try {
    var sheet = getMasterSheet("ConfigMaestro");
    var data = sheet.getDataRange().getValues();
    var config = { wa_soporte: WA_SOPORTE_DEFAULT };
    for (var i = 1; i < data.length; i++) {
      if (data[i][0]) config[String(data[i][0])] = String(data[i][1]);
    }
    return { ok: true, config: config };
  } catch(e) {
    return { ok: true, config: { wa_soporte: WA_SOPORTE_DEFAULT } };
  }
}

function guardarConfigMaestroFn(clave, valor) {
  try {
    var sheet = getMasterSheet("ConfigMaestro");
    // Asegurar encabezados
    if (sheet.getLastRow() === 0) sheet.appendRow(['clave', 'valor']);
    var data = sheet.getDataRange().getValues();
    var idx = -1;
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(clave)) { idx = i + 1; break; }
    }
    if (idx > 0) sheet.getRange(idx, 2).setValue(valor);
    else sheet.appendRow([clave, valor]);
    return { ok: true };
  } catch(e) {
    return { ok: false, error: e.toString() };
  }
}

function obtenerClientes() {
  try {
    var sheet = getMasterSheet("Clientes");
    var data = sheet.getDataRange().getValues();
    var h = data.shift();
    return data.map(r => {
      return {
        ID_Sistema: r[0],
        Nombre_Empresa: r[1],
        Correo: r[2],
        ID_BaseDatos: r[4],
        Plan: r[5],
        Estatus: r[6],
        Fecha_Registro: r[7],
        Fecha_Vencimiento: r[8]
      };
    });
  } catch(e) {
    return [];
  }
}

function obtenerPlanes() {
  try {
    var sheet = getMasterSheet("Planes");
    if (sheet.getLastRow() < 2) {
      return [
        { nombre: "BASIC", dias: 30, precio: 10 },
        { nombre: "PRO", dias: 30, precio: 20 },
        { nombre: "PREMIUM", dias: 365, precio: 200 }
      ];
    }
    var data = sheet.getDataRange().getValues();
    data.shift(); // Quitar cabecera
    return data.map(r => ({ nombre: r[0], dias: r[1], precio: r[2] }));
  } catch(e) {
    return [];
  }
}

function actualizarPlanes(planes) {
  try {
    var sheet = getMasterSheet("Planes");
    sheet.clear();
    sheet.appendRow(["nombre", "dias", "precio"]);
    if (planes && planes.length > 0) {
      planes.forEach(p => {
        sheet.appendRow([p.nombre, p.dias, p.precio]);
      });
    }
    return { ok: true };
  } catch(e) {
    return { ok: false, error: e.toString() };
  }
}
